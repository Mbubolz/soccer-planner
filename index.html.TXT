<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Soccer Session & Template Builder</title>
    <script src="https://cdn.tailwindcss.com"></script>
    <script src="https://unpkg.com/react@18/umd/react.development.js" crossorigin></script>
    <script src="https://unpkg.com/react-dom@18/umd/react-dom.development.js" crossorigin></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body class="bg-slate-100 text-slate-800">
    <div id="root"></div>

    <script type="text/babel">
        function App() {
            // Load saved templates or default ones
            const [templates, setTemplates] = React.useState([
                { id: 'pro', name: 'Pro Session Planner (4-Activity Grid)', type: 'pro' },
                { id: 'grid', name: 'Weekly Matrix Notebook Grid', type: 'grid' }
            ]);

            const [activeTemplateId, setActiveTemplateId] = React.useState('pro');
            const [showNewTemplateModal, setShowNewTemplateModal] = React.useState(false);
            const [newTemplateName, setNewTemplateName] = React.useState('');
            const [newTemplateType, setNewTemplateType] = React.useState('pro');

            // Persistent Session Data using localStorage
            const [sessionData, setSessionData] = React.useState(() => {
                const saved = localStorage.getItem('soccer_session_data');
                return saved ? JSON.parse(saved) : {
                    team: 'Alpha Futbol',
                    date: new Date().toISOString().split('T')[0],
                    theme: 'Positional Play & Transition',
                    duration: '90 Minutes',
                    activity1Name: 'Warm-up / Activation',
                    activity1Desc: 'Passing patterns focused on third-man runs.',
                    activity2Name: 'Small-Sided Game (3v3 + 2 neutrals)',
                    activity2Desc: 'Exploit half-spaces and maintain possession.'
                };
            });

            React.useEffect(() => {
                localStorage.setItem('soccer_session_data', JSON.stringify(sessionData));
            }, [sessionData]);

            const handleChange = (e) => {
                setSessionData({ ...sessionData, [e.target.name]: e.target.value });
            };

            const handleAddTemplate = (e) => {
                e.preventDefault();
                if (!newTemplateName.trim()) return;
                const newId = 'custom_' + Date.now();
                const newTpl = { id: newId, name: newTemplateName, type: newTemplateType };
                setTemplates([...templates, newTpl]);
                setActiveTemplateId(newId);
                setNewTemplateName('');
                setShowNewTemplateModal(false);
            };

            const activeTpl = templates.find(t => t.id === activeTemplateId) || templates[0];

            return (
                <div className="max-w-4xl mx-auto p-6">
                    {/* Control Bar */}
                    <div className="flex flex-wrap justify-between items-center mb-6 bg-white p-4 rounded-lg shadow print:hidden gap-4">
                        <div className="flex items-center gap-3">
                            <div>
                                <label className="block text-xs font-bold uppercase text-slate-500 mb-1">Select Template</label>
                                <select 
                                    value={activeTemplateId} 
                                    onChange={(e) => setActiveTemplateId(e.target.value)}
                                    className="border border-slate-300 rounded p-2 text-sm bg-slate-50"
                                >
                                    {templates.map(tpl => (
                                        <option key={tpl.id} value={tpl.id}>{tpl.name}</option>
                                    ))}
                                </select>
                            </div>
                            <button 
                                onClick={() => setShowNewTemplateModal(true)}
                                className="mt-5 bg-slate-800 hover:bg-slate-900 text-white px-3 py-2 rounded text-sm transition font-medium"
                            >
                                + New Template
                            </button>
                        </div>
                        <button 
                            onClick={() => window.print()}
                            className="mt-5 bg-emerald-600 hover:bg-emerald-700 text-white font-medium px-4 py-2 rounded text-sm transition"
                        >
                            Print / Save to PDF
                        </button>
                    </div>

                    {/* New Template Modal */}
                    {showNewTemplateModal && (
                        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50 print:hidden">
                            <form onSubmit={handleAddTemplate} className="bg-white p-6 rounded-lg max-w-md w-full shadow-xl">
                                <h3 className="font-bold text-lg mb-4 text-slate-800">Create Custom Template</h3>
                                <label className="block text-xs font-bold uppercase text-slate-500 mb-1">Template Name</label>
                                <input 
                                    type="text" 
                                    placeholder="e.g., Matchday Minus 2" 
                                    value={newTemplateName}
                                    onChange={(e) => setNewTemplateName(e.target.value)}
                                    className="w-full border border-slate-300 rounded p-2 mb-4 text-sm"
                                    required
                                />
                                <label className="block text-xs font-bold uppercase text-slate-500 mb-1">Base Layout Structure</label>
                                <select 
                                    value={newTemplateType}
                                    onChange={(e) => setNewTemplateType(e.target.value)}
                                    className="w-full border border-slate-300 rounded p-2 mb-6 text-sm bg-slate-50"
                                >
                                    <option value="pro">Pro Session Planner (Activity Cards)</option>
                                    <option value="grid">Weekly Matrix Notebook Grid</option>
                                </select>
                                <div className="flex justify-end gap-2">
                                    <button 
                                        type="button" 
                                        onClick={() => setShowNewTemplateModal(false)}
                                        className="px-4 py-2 text-sm text-slate-600 hover:bg-slate-100 rounded"
                                    >
                                        Cancel
                                    </button>
                                    <button 
                                        type="submit" 
                                        className="px-4 py-2 text-sm bg-emerald-600 hover:bg-emerald-700 text-white rounded font-medium"
                                    >
                                        Save Template
                                    </button>
                                </div>
                            </form>
                        </div>
                    )}

                    {/* Dynamic Template Container */}
                    <div className="bg-white border border-slate-300 shadow-lg p-6 rounded-none print:shadow-none print:border-none">
                        {activeTpl.type === 'pro' ? (
                            <div>
                                <div className="border border-slate-800 mb-4 bg-slate-900 text-white p-3 text-center font-bold tracking-wide uppercase text-sm">
                                    {activeTpl.name} — Technical Session Plan
                                </div>

                                <div className="grid grid-cols-4 gap-2 mb-4 text-xs">
                                    <div className="border border-slate-400 p-2">
                                        <span className="block font-bold text-slate-500">TEAM</span>
                                        <input type="text" name="team" value={sessionData.team} onChange={handleChange} className="w-full font-semibold bg-transparent focus:outline-none" />
                                    </div>
                                    <div className="border border-slate-400 p-2">
                                        <span className="block font-bold text-slate-500">DATE</span>
                                        <input type="date" name="date" value={sessionData.date} onChange={handleChange} className="w-full font-semibold bg-transparent focus:outline-none" />
                                    </div>
                                    <div className="border border-slate-400 p-2">
                                        <span className="block font-bold text-slate-500">THEME</span>
                                        <input type="text" name="theme" value={sessionData.theme} onChange={handleChange} className="w-full font-semibold bg-transparent focus:outline-none" />
                                    </div>
                                    <div className="border border-slate-400 p-2">
                                        <span className="block font-bold text-slate-500">DURATION</span>
                                        <input type="text" name="duration" value={sessionData.duration} onChange={handleChange} className="w-full font-semibold bg-transparent focus:outline-none" />
                                    </div>
                                </div>

                                <div className="grid grid-cols-2 gap-4">
                                    <div className="border border-slate-400 p-3">
                                        <input type="text" name="activity1Name" value={sessionData.activity1Name} onChange={handleChange} className="w-full font-bold text-sm bg-slate-100 p-1 mb-2 border border-slate-300" />
                                        <textarea name="activity1Desc" value={sessionData.activity1Desc} onChange={handleChange} className="w-full text-xs bg-transparent border border-slate-200 p-1 h-24 resize-none focus:outline-none"></textarea>
                                    </div>
                                    <div className="border border-slate-400 p-3">
                                        <input type="text" name="activity2Name" value={sessionData.activity2Name} onChange={handleChange} className="w-full font-bold text-sm bg-slate-100 p-1 mb-2 border border-slate-300" />
                                        <textarea name="activity2Desc" value={sessionData.activity2Desc} onChange={handleChange} className="w-full text-xs bg-transparent border border-slate-200 p-1 h-24 resize-none focus:outline-none"></textarea>
                                    </div>
                                </div>
                            </div>
                        ) : (
                            <div>
                                <div className="border-b-2 border-slate-800 pb-2 mb-4 font-bold text-lg text-slate-700">
                                    {activeTpl.name}
                                </div>
                                <table className="w-full border-collapse border border-slate-400 text-xs">
                                    <thead>
                                        <tr className="bg-slate-200">
                                            <th className="border border-slate-400 p-2">Day</th>
                                            <th className="border border-slate-400 p-2">Exercise / Activity</th>
                                            <th className="border border-slate-400 p-2">Repetition / Duration</th>
                                        </tr>
                                    </thead>
                                    <tbody>
                                        {['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'].map((day, idx) => (
                                            <tr key={idx}>
                                                <td className="border border-slate-400 p-2 font-bold bg-slate-50 w-1/4">{day}</td>
                                                <td className="border border-slate-400 p-2"><input type="text" placeholder="Enter activity..." className="w-full bg-transparent focus:outline-none" /></td>
                                                <td className="border border-slate-400 p-2"><input type="text" placeholder="e.g. 4x4 mins" className="w-full bg-transparent focus:outline-none" /></td>
                                            </tr>
                                        ))}
                                    </tbody>
                                </table>
                            </div>
                        )}
                    </div>
                </div>
            );
        }

        ReactDOM.render(<App />, document.getElementById('root'));
    </script>
</body>
</html>
